 522:                 else if (logSignCount >= this._minCloseSign)
 523:                     return TradeSignal.CloseLong;
 524:                 else
 525:                     return TradeSignal.Wait;
 526:             else
 527:                 return TradeSignal.Unknown;

 532:         private TradeSignal CalculateTradeSignal2()
 533:         {
 534:             int longCount = 0, shortCount = 0;
 535:             int longExitCount = 0, shortExitCount = 0;
 536: 
 537:             double rvol = GetLineValueByName(this._atrIndicator, "RvolSignal");
 538:             double hma = GetLineValueByName(this._atrIndicator, "HMA_Direction");
 539:             double vdps = GetLineValueByName(this._deltaBaseIndicator, "APAVD_Flag");
 540:             double vdStrong = GetLineValueByName(this._deltaBaseIndicator, "VD_Strength_Flag");
 541:             double vdp = GetLineValueByName(this._deltaBaseIndicator, "VD_Price_Divergent_Flag");
 542:             double vdtv = GetLineValueByName(this._deltaBaseIndicator, "VD_to_Volume_Flag");
 543: 
 544:             void Tally(bool use, double v, ref int pos, ref int neg)
 545:             {
 546:                 if (!use) return;
 547:                 if (v > 0) pos++;
 548:                 else if (v < 0) neg++;
 549:             }
 550: 
 551:             // Entry tally
 552:             Tally(_entryUseRVOL, rvol, ref longCount, ref shortCount);
 553:             Tally(_entryUseHMA, hma, ref longCount, ref shortCount);
 554:             Tally(_entryUseVDPS, vdps, ref longCount, ref shortCount);
 555:             Tally(_entryUseVDStrong, vdStrong, ref longCount, ref shortCount);
 556:             Tally(_entryUseVDtV, vdtv, ref longCount, ref shortCount);
 557:             Tally(_entryUseVDP, vdp, ref longCount, ref shortCount);
 558: 
 559:             // Exit tally
 560:             Tally(_exitUseRVOL, rvol, ref longExitCount, ref shortExitCount);
 561:             Tally(_exitUseHMA, hma, ref longExitCount, ref shortExitCount);
 562:             Tally(_exitUseVDPS, vdps, ref longExitCount, ref shortExitCount);
 563:             Tally(_exitUseVDStrong, vdStrong, ref longExitCount, ref shortExitCount);
 564:             Tally(_exitUseVDtV, vdtv, ref longExitCount, ref shortExitCount);
 565:             Tally(_exitUseVDP, vdp, ref longExitCount, ref shortExitCount);
 566: 
 567:             if (longCount == shortCount)
 568:                 return TradeSignal.Unknown;
 569:             else if (longCount > shortCount)
 570:                 if (longCount >= Math.Max(this._minTradeSign, _entryMinConditions))
 571:                     return TradeSignal.OpenBuy;
 572:                 else if (longExitCount >= Math.Max(this._minCloseSign, _exitMinConditions))
 573:                     return TradeSignal.CloseSell;
 574:                 else
 575:                     return TradeSignal.Wait;
 576:             else if (shortCount > longCount)
 577:                 if (shortCount >= Math.Max(this._minTradeSign, _entryMinConditions))
 578:                     return TradeSignal.OpenSell;
 579:                 else if (shortExitCount >= Math.Max(this._minCloseSign, _exitMinConditions))
 580:                     return TradeSignal.CloseLong;
 581:                 else
 582:                     return TradeSignal.Wait;
 583:             else
